cmake_minimum_required(VERSION 3.20)
project(jnative_runtime_tests LANGUAGES C CXX)
include(src/main/cpp/jn_features.cmake)
find_package(Threads REQUIRED)
include(src/main/cpp/jn_platform.cmake)
add_library(jnative_runtime src/main/cpp/jn_runtime.cpp src/main/cpp/jn_arrays.cpp src/main/cpp/jn_string_kernels.cpp src/main/cpp/jn_classlib_numbers.cpp src/main/cpp/jn_numbers.cpp src/main/cpp/jn_codecs.cpp src/main/cpp/jn_threads.cpp src/main/cpp/jn_abi.cpp src/main/cpp/jn_files.cpp src/main/cpp/jn_buffers.cpp src/main/cpp/jn_reflection.cpp src/main/cpp/jn_console.cpp src/main/cpp/jn_diagnostics_none.cpp)
target_include_directories(jnative_runtime PUBLIC src/main/cpp)
target_link_libraries(jnative_runtime PUBLIC jnative_features jnative_platform Threads::Threads ${CMAKE_DL_LIBS})
if(MSVC)
    target_compile_options(jnative_runtime PRIVATE /fp:strict /W4)
else()
    target_compile_options(jnative_runtime PRIVATE -fno-fast-math -ffp-contract=off -Wall -Wextra -Wpedantic)
endif()
add_executable(runtime_tests src/test/cpp/runtime_tests.cpp)
target_link_libraries(runtime_tests PRIVATE jnative_runtime)
enable_testing()
add_test(NAME runtime_contracts COMMAND runtime_tests)
option(JN_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(JN_SANITIZERS AND NOT MSVC)
    target_compile_options(jnative_platform PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_compile_options(jnative_runtime PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(jnative_runtime PUBLIC -fsanitize=address,undefined)
endif()
