cmake_minimum_required(VERSION 3.20)
project(jnative_runtime_tests LANGUAGES C CXX)
include(src/main/cpp/jn_features.cmake)
find_package(Threads REQUIRED)
include(src/main/cpp/jn_platform.cmake)
add_library(jnative_runtime src/main/cpp/jn_allocator.cpp src/main/cpp/jn_runtime.cpp src/main/cpp/jn_arrays.cpp src/main/cpp/jn_string_kernels.cpp src/main/cpp/jn_classlib_numbers.cpp src/main/cpp/jn_numbers.cpp src/main/cpp/jn_codecs.cpp src/main/cpp/jn_threads.cpp src/main/cpp/jn_abi.cpp src/main/cpp/jn_files.cpp src/main/cpp/jn_buffers.cpp src/main/cpp/jn_reflection.cpp src/main/cpp/jn_console.cpp src/main/cpp/jn_diagnostics_none.cpp)
target_include_directories(jnative_runtime PUBLIC src/main/cpp)
target_link_libraries(jnative_runtime PUBLIC jnative_features jnative_platform Threads::Threads ${CMAKE_DL_LIBS})
if(MSVC)
    target_compile_options(jnative_runtime PRIVATE /fp:strict /W4)
else()
    target_compile_options(jnative_runtime PRIVATE -fno-fast-math -ffp-contract=off -Wall -Wextra -Wpedantic)
endif()
add_executable(runtime_tests src/test/cpp/runtime_tests.cpp)
target_link_libraries(runtime_tests PRIVATE jnative_runtime)
enable_testing()
add_test(NAME runtime_contracts COMMAND runtime_tests)
add_executable(primitive_array_allocation_tests src/test/cpp/primitive_array_allocation_tests.cpp)
target_link_libraries(primitive_array_allocation_tests PRIVATE jnative_runtime)
add_test(NAME primitive_array_allocation_contracts COMMAND primitive_array_allocation_tests)
add_executable(allocation_context_tests src/test/cpp/allocation_context_tests.cpp)
target_link_libraries(allocation_context_tests PRIVATE jnative_runtime)
add_test(NAME allocation_context_shells COMMAND allocation_context_tests shells)
add_test(NAME allocation_context_arrays COMMAND allocation_context_tests arrays)
add_test(NAME allocation_context_failures COMMAND allocation_context_tests failures)
add_test(NAME allocation_context_pressure COMMAND allocation_context_tests pressure)
add_test(NAME allocation_context_concurrent_pressure COMMAND allocation_context_tests concurrent-pressure)
add_test(NAME allocation_context_automatic_pressure COMMAND allocation_context_tests automatic-pressure)
add_test(NAME allocation_context_byte_pressure COMMAND allocation_context_tests byte-pressure)
add_test(NAME allocation_context_byte_detach COMMAND allocation_context_tests byte-detach)
add_test(NAME allocation_context_byte_survivors COMMAND allocation_context_tests byte-survivors)
add_test(NAME allocation_context_cache_budget COMMAND allocation_context_tests cache-budget)
add_test(NAME allocation_context_arena_ownership COMMAND allocation_context_tests arena-ownership)
add_executable(runtime_isolation_benchmark EXCLUDE_FROM_ALL src/test/cpp/runtime_isolation_benchmark.cpp)
target_link_libraries(runtime_isolation_benchmark PRIVATE jnative_runtime)
add_library(transfer_code_probe OBJECT EXCLUDE_FROM_ALL src/test/cpp/transfer_code_probe.cpp)
target_link_libraries(transfer_code_probe PRIVATE jnative_runtime)
if(MSVC)
    target_compile_options(transfer_code_probe PRIVATE /O2 /fp:strict /FAs "/Fa${CMAKE_CURRENT_BINARY_DIR}/transfer_code_probe.asm")
endif()
option(JN_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(JN_SANITIZERS AND NOT MSVC)
    target_compile_options(jnative_platform PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_compile_options(jnative_runtime PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(jnative_runtime PUBLIC -fsanitize=address,undefined)
endif()
