cmake_minimum_required(VERSION 3.20)
project(portability_host LANGUAGES C CXX)
if(NOT JNATIVE_PROJECT)
    message(FATAL_ERROR "Set JNATIVE_PROJECT to the generated native project directory")
endif()
set(JNATIVE_OUTPUT_KIND STATIC_LIBRARY)
add_subdirectory("${JNATIVE_PROJECT}" "${CMAKE_CURRENT_BINARY_DIR}/application")
add_executable(portability_host host.cpp)
target_link_libraries(portability_host PRIVATE jNative::Application)
if(MINGW)
    target_link_options(portability_host PRIVATE -static-libgcc -static-libstdc++)
endif()
